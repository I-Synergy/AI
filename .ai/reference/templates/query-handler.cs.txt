/// Query to retrieve one {entity} by its identifier.

/// <summary>

public sealed record Get{Entity}ByIdResponse(Models.{Entity}? {Entity});

using {ApplicationName}.Data.Abstractions;
using ISynergy.Framework.CQRS.Queries;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;

namespace {ApplicationName}.Domains.Features.{Entities}.Queries.Get{Entity}ById;

/// <summary>
/// Handles retrieving the {entity} by its identifier.
/// </summary>
public sealed class Get{Entity}ByIdQueryHandler(
    IDataContext dataContext,
    ILogger<Get{Entity}ByIdQueryHandler> logger)
    : IQueryHandler<Get{Entity}ByIdQuery, Get{Entity}ByIdResponse>
{
    /// <inheritdoc />
    public async Task<Get{Entity}ByIdResponse> HandleAsync(
        Get{Entity}ByIdQuery query,
        CancellationToken cancellationToken = default)
    {
        logger.LogDebug("Retrieving {Entity} {EntityId}", query.{Entity}Id);

        var model = await dataContext.{Entities}
            .Where(e => e.{Entity}Id == query.{Entity}Id)
            .Select(e => new Models.{Entity}(/* map e to the model constructor, property by property */))
            .FirstOrDefaultAsync(cancellationToken);

        return new Get{Entity}ByIdResponse(model);
    }
}

public sealed record Get{Entities}ListResponse(IReadOnlyList<Models.{Entity}> {Entities});


// File: {ApplicationName}.Domains/Features/{Entities}/Queries/Get{Entities}List/Get{Entities}ListQueryHandler.cs

using {ApplicationName}.Data.Abstractions;
using ISynergy.Framework.CQRS.Queries;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;

namespace {ApplicationName}.Domains.Features.{Entities}.Queries.Get{Entities}List;

/// <summary>
/// Handles retrieving all {entities}.
/// </summary>
public sealed class Get{Entities}ListQueryHandler(
    IDataContext dataContext,
    ILogger<Get{Entities}ListQueryHandler> logger)
    : IQueryHandler<Get{Entities}ListQuery, Get{Entities}ListResponse>
{
    /// <inheritdoc />
    public async Task<Get{Entities}ListResponse> HandleAsync(
        Get{Entities}ListQuery query,
        CancellationToken cancellationToken = default)
    {
        logger.LogDebug("Retrieving all {entities}");

        var models = await dataContext.{Entities}
            .OrderBy(e => e.Name)
            .Select(e => new Models.{Entity}(/* map e to the model constructor, property by property */))
            .ToListAsync(cancellationToken);

        logger.LogDebug("Retrieved {Count} {entities}", models.Count);

        return new Get{Entities}ListResponse(models);
    }
}
