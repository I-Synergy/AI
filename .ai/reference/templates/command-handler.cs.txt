using {ApplicationName}.Data.Abstractions;
using ISynergy.Framework.CQRS.Abstractions.Commands;
using Microsoft.Extensions.Logging;

namespace {ApplicationName}.Domains.Features.{Entities}.Commands.Create{Entity};

/// <summary>
/// Handles the creation of the {entity}.
/// </summary>
public sealed class Create{Entity}CommandHandler(
    IDataContext dataContext,
    ILogger<Create{Entity}CommandHandler> logger)
    : ICommandHandler<Create{Entity}Command, Create{Entity}Response>
{
    /// <inheritdoc />
    public async Task<Create{Entity}Response> HandleAsync(
        Create{Entity}Command command,
        CancellationToken cancellationToken = default)
    {
        var entity = new Entities.{Entity}
        {
            {Entity}Id = Guid.NewGuid(),
            Name = command.Name,
            Description = command.Description,
            CreatedAt = DateTimeOffset.UtcNow
        };

        dataContext.{Entities}.Add(entity);
        var rowsAffected = await dataContext.SaveChangesAsync(cancellationToken);

        if (rowsAffected == 0)
        {
            logger.LogWarning("Creating the {Entity} wrote no rows");
            throw new InvalidOperationException("The {entity} could not be created.");
        }

        logger.LogInformation("Created {Entity} {EntityId}", entity.{Entity}Id);
        return new Create{Entity}Response(entity.{Entity}Id);
    }
}

using {ApplicationName}.Data.Abstractions;
using ISynergy.Framework.CQRS.Abstractions.Commands;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;

namespace {ApplicationName}.Domains.Features.{Entities}.Commands.Update{Entity};

/// <summary>
/// Handles updating the {entity}.
/// </summary>
public sealed class Update{Entity}CommandHandler(
    IDataContext dataContext,
    ILogger<Update{Entity}CommandHandler> logger)
    : ICommandHandler<Update{Entity}Command, Update{Entity}Response>
{
    /// <inheritdoc />
    public async Task<Update{Entity}Response> HandleAsync(
        Update{Entity}Command command,
        CancellationToken cancellationToken = default)
    {
        var entity = await dataContext.{Entities}.FirstOrDefaultAsync(e => e.{Entity}Id == command.{Entity}Id, cancellationToken);

        if (entity is null)
        {
            logger.LogWarning("{Entity} {EntityId} was not found for update", command.{Entity}Id);
            throw new ResourceNotFoundException("The {entity} was not found.");
        }

        // The change tracker detects the mutations; no Update() call is needed. SaveChanges reports zero rows
        // when the values did not change, which is not a failure, so the row count is not checked here.
        entity.Name = command.Name;
        entity.Description = command.Description;
        entity.Status = command.Status;

        await dataContext.SaveChangesAsync(cancellationToken);

        logger.LogInformation("Updated {Entity} {EntityId}", command.{Entity}Id);
        return new Update{Entity}Response(true);
    }
}

using {ApplicationName}.Data.Abstractions;
using ISynergy.Framework.CQRS.Abstractions.Commands;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;

namespace {ApplicationName}.Domains.Features.{Entities}.Commands.Delete{Entity};

/// <summary>
/// Handles deleting the {entity}.
/// </summary>
public sealed class Delete{Entity}CommandHandler(
    IDataContext dataContext,
    ILogger<Delete{Entity}CommandHandler> logger)
    : ICommandHandler<Delete{Entity}Command, Delete{Entity}Response>
{
    /// <inheritdoc />
    public async Task<Delete{Entity}Response> HandleAsync(
        Delete{Entity}Command command,
        CancellationToken cancellationToken = default)
    {
        var entity = await dataContext.{Entities}.FirstOrDefaultAsync(e => e.{Entity}Id == command.{Entity}Id, cancellationToken);

        if (entity is null)
        {
            logger.LogWarning("{Entity} {EntityId} was not found for deletion", command.{Entity}Id);
            throw new ResourceNotFoundException("The {entity} was not found.");
        }

        dataContext.{Entities}.Remove(entity);
        var rowsAffected = await dataContext.SaveChangesAsync(cancellationToken);

        if (rowsAffected == 0)
        {
            logger.LogWarning("Deleting {Entity} {EntityId} wrote no rows", command.{Entity}Id);
            throw new InvalidOperationException("The {entity} could not be deleted.");
        }

        logger.LogInformation("Deleted {Entity} {EntityId}", command.{Entity}Id);
        return new Delete{Entity}Response(true);
    }
}
