using System.Net;
using System.Net.Http.Json;
using System.Text.Json;
using Microsoft.VisualStudio.TestTools.UnitTesting;

namespace {ApplicationName}.Service.Tests;

/// <summary>
/// Exercises the {Entities} endpoints end to end through the real pipeline: routing, validation, the
/// handlers and an in-memory database.
/// </summary>
/// <remarks>
/// Request bodies are anonymous objects and responses are read as JSON, so these tests pin the wire
/// contract rather than the C# types behind it. Traces: US-&lt;Domain&gt;-NNN (placeholder; tag methods with
/// <c>[TestCategory("AC-&lt;Domain&gt;-NNN.n")]</c> once the criteria exist, see .ai/reference/traceability.md).
/// </remarks>
[TestClass]
public sealed class {Entities}EndpointsTests
{
    [TestMethod]
    public async Task Create_ValidRequest_ThenGetById_ReturnsTheCreated{Entity}()
    {
        // Arrange
        await using var factory = new CustomWebApplicationFactory();
        using var client = factory.CreateClient();

        // Act: create
        using var createResponse = await client.PostAsJsonAsync("/{entities}", new
        {
            name = "Alpha",
            description = "First sample"
        });
        var createBody = await createResponse.Content.ReadAsStringAsync();

        Assert.AreEqual(HttpStatusCode.Created, createResponse.StatusCode, $"Unexpected create status. Body: {createBody}");

        var id = JsonSerializer.Deserialize<Guid>(createBody);
        Assert.AreNotEqual(Guid.Empty, id, "The created identifier was empty.");

        // Act: get by id
        using var getResponse = await client.GetAsync($"/{entities}/{id}");
        var getBody = await getResponse.Content.ReadAsStringAsync();

        // Assert
        Assert.AreEqual(HttpStatusCode.OK, getResponse.StatusCode, $"Unexpected get status. Body: {getBody}");

        var stored = JsonSerializer.Deserialize<JsonElement>(getBody);
        Assert.AreEqual("Alpha", stored.GetProperty("name").GetString(), "The name did not round-trip.");
        Assert.AreEqual("First sample", stored.GetProperty("description").GetString(), "The description did not round-trip.");
    }

    [TestMethod]
    public async Task Create_ValidRequest_ThenList_IncludesTheCreated{Entity}()
    {
        // Arrange
        await using var factory = new CustomWebApplicationFactory();
        using var client = factory.CreateClient();

        using var createResponse = await client.PostAsJsonAsync("/{entities}", new { name = "Listed", description = "In the list" });
        var id = JsonSerializer.Deserialize<Guid>(await createResponse.Content.ReadAsStringAsync());

        // Act
        using var listResponse = await client.GetAsync("/{entities}");
        var listBody = await listResponse.Content.ReadAsStringAsync();

        // Assert
        Assert.AreEqual(HttpStatusCode.OK, listResponse.StatusCode, $"Unexpected list status. Body: {listBody}");

        var items = JsonSerializer.Deserialize<JsonElement>(listBody);
        Assert.AreEqual(JsonValueKind.Array, items.ValueKind, "The list endpoint did not return an array.");
        Assert.IsTrue(
            items.EnumerateArray().Any(row => row.GetProperty("name").GetString() == "Listed"),
            $"The created row {id} was not in the list: {listBody}");
    }

    [TestMethod]
    public async Task GetById_UnknownId_ReturnsNotFound()
    {
        // Arrange
        await using var factory = new CustomWebApplicationFactory();
        using var client = factory.CreateClient();

        // Act
        using var response = await client.GetAsync($"/{entities}/{Guid.NewGuid()}");

        // Assert
        Assert.AreEqual(HttpStatusCode.NotFound, response.StatusCode);
    }

    [TestMethod]
    public async Task Create_WithoutAName_IsRefusedWithAValidationProblem()
    {
        // Arrange
        await using var factory = new CustomWebApplicationFactory();
        using var client = factory.CreateClient();

        // Act
        using var response = await client.PostAsJsonAsync("/{entities}", new { description = "No name" });

        // Assert
        Assert.AreEqual(HttpStatusCode.BadRequest, response.StatusCode);
    }

    [TestMethod]
    public async Task Delete_Existing{Entity}_ThenGetById_ReturnsNotFound()
    {
        // Arrange
        await using var factory = new CustomWebApplicationFactory();
        using var client = factory.CreateClient();

        using var createResponse = await client.PostAsJsonAsync("/{entities}", new { name = "Doomed", description = "Short lived" });
        var id = JsonSerializer.Deserialize<Guid>(await createResponse.Content.ReadAsStringAsync());

        // Act
        using var deleteResponse = await client.DeleteAsync($"/{entities}/{id}");
        using var getResponse = await client.GetAsync($"/{entities}/{id}");

        // Assert
        Assert.AreEqual(HttpStatusCode.NoContent, deleteResponse.StatusCode);
        Assert.AreEqual(HttpStatusCode.NotFound, getResponse.StatusCode);
    }
}
