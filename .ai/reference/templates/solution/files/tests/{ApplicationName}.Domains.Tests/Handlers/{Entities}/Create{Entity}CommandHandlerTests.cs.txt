using {ApplicationName}.Domains.Tests.Infrastructure;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging.Abstractions;
using Microsoft.VisualStudio.TestTools.UnitTesting;

namespace {ApplicationName}.Domains.Tests.Handlers.{Entities};

/// <summary>
/// Verifies that the create command persists the given values through the real EF Core provider.
/// </summary>
[TestClass]
public sealed class Create{Entity}CommandHandlerTests
{
    [TestMethod]
    [DataRow("Alpha", "First sample")]
    [DataRow("Beta", null)]
    public async Task HandleAsync_ValidCommand_PersistsTheGivenValues(string name, string? description)
    {
        // Arrange
        using var db = new TestDatabase();
        var handler = new Create{Entity}CommandHandler(
            db.Context,
            NullLogger<Create{Entity}CommandHandler>.Instance);

        var command = new Create{Entity}Command(Name: name, Description: description);

        // Act
        var result = await handler.HandleAsync(command);

        // Assert
        Assert.IsNotNull(result);
        Assert.AreNotEqual(Guid.Empty, result.{Entity}Id);

        var stored = await db.Context.{Entities}.SingleAsync(e => e.{Entity}Id == result.{Entity}Id);
        Assert.AreEqual(name, stored.Name);
        Assert.AreEqual(description, stored.Description);
        Assert.AreNotEqual(default, stored.CreatedAt);
    }

    [TestMethod]
    public async Task HandleAsync_TwoCommands_PersistTwoRowsWithDistinctIds()
    {
        // Arrange
        using var db = new TestDatabase();
        var handler = new Create{Entity}CommandHandler(
            db.Context,
            NullLogger<Create{Entity}CommandHandler>.Instance);

        // Act
        var first = await handler.HandleAsync(new Create{Entity}Command(Name: "One", Description: null));
        var second = await handler.HandleAsync(new Create{Entity}Command(Name: "Two", Description: null));

        // Assert
        Assert.AreNotEqual(first.{Entity}Id, second.{Entity}Id);
        Assert.AreEqual(2, await db.Context.{Entities}.CountAsync());
    }
}
