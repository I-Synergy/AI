using {ApplicationName}.Domains;
using {ApplicationName}.Domains.Features.{Entities}.Commands.Delete{Entity};
using {ApplicationName}.Domains.Tests.Infrastructure;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging.Abstractions;
using Microsoft.VisualStudio.TestTools.UnitTesting;

namespace {ApplicationName}.Domains.Tests.Handlers.{Entities};

// Traces: US-<Domain>-NNN (placeholder; fill in when the story exists, see .ai/reference/traceability.md).
// Tag each test method with [TestCategory("AC-<Domain>-NNN.n")] once its acceptance criterion is declared.
/// <summary>
/// Verifies that the delete command removes the row and refuses a missing row.
/// </summary>
[TestClass]
public sealed class Delete{Entity}CommandHandlerTests
{
    [TestMethod]
    public async Task HandleAsync_Existing{Entity}_RemovesTheRowAndReportsSuccess()
    {
        // Arrange
        using var db = new TestDatabase();
        var id = await {Entity}TestData.AddAsync(db.Context);
        var handler = new Delete{Entity}CommandHandler(
            db.Context,
            NullLogger<Delete{Entity}CommandHandler>.Instance);

        // Act
        var result = await handler.HandleAsync(new Delete{Entity}Command(id));

        // Assert
        Assert.IsTrue(result.Success);
        Assert.IsFalse(await db.Context.{Entities}.AnyAsync(e => e.{Entity}Id == id));
    }

    [TestMethod]
    public async Task HandleAsync_Existing{Entity}_LeavesOtherRowsAlone()
    {
        // Arrange
        using var db = new TestDatabase();
        var id = await {Entity}TestData.AddAsync(db.Context, name: "Target");
        var otherId = await {Entity}TestData.AddAsync(db.Context, name: "Bystander");
        var handler = new Delete{Entity}CommandHandler(
            db.Context,
            NullLogger<Delete{Entity}CommandHandler>.Instance);

        // Act
        await handler.HandleAsync(new Delete{Entity}Command(id));

        // Assert
        Assert.IsTrue(await db.Context.{Entities}.AnyAsync(e => e.{Entity}Id == otherId));
    }

    [TestMethod]
    public async Task HandleAsync_Missing{Entity}_ThrowsResourceNotFoundException()
    {
        // Arrange
        using var db = new TestDatabase();
        var handler = new Delete{Entity}CommandHandler(
            db.Context,
            NullLogger<Delete{Entity}CommandHandler>.Instance);

        var command = new Delete{Entity}Command(Guid.NewGuid());

        // Act & Assert
        await Assert.ThrowsExactlyAsync<ResourceNotFoundException>(() => handler.HandleAsync(command));
    }
}
