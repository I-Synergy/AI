/// <summary>
/// Verifies that the by-identifier query returns the stored row or nothing.
/// </summary>
[TestClass]
public sealed class Get{Entity}ByIdQueryHandlerTests
{
    [TestMethod]
    public async Task HandleAsync_Existing{Entity}_ReturnsTheMappedModel()
    {
        // Arrange
        using var db = new TestDatabase();
        var id = await {Entity}TestData.AddAsync(
            db.Context,
            name: "Alpha",
            description: "First sample",
            status: {Entity}Status.Active);
        var handler = new Get{Entity}ByIdQueryHandler(
            db.Context,
            NullLogger<Get{Entity}ByIdQueryHandler>.Instance);

        // Act
        var result = await handler.HandleAsync(new Get{Entity}ByIdQuery(id));

        // Assert
        Assert.IsNotNull(result.{Entity});
        Assert.AreEqual(id, result.{Entity}.{Entity}Id);
        Assert.AreEqual("Alpha", result.{Entity}.Name);
        Assert.AreEqual("First sample", result.{Entity}.Description);
        Assert.AreEqual({Entity}Status.Active, result.{Entity}.Status);
    }

    [TestMethod]
    public async Task HandleAsync_Missing{Entity}_ReturnsANullModel()
    {
        // Arrange
        using var db = new TestDatabase();
        var handler = new Get{Entity}ByIdQueryHandler(
            db.Context,
            NullLogger<Get{Entity}ByIdQueryHandler>.Instance);

        // Act
        var result = await handler.HandleAsync(new Get{Entity}ByIdQuery(Guid.NewGuid()));

        // Assert
        Assert.IsNull(result.{Entity});
    }
}
