using System.Reflection;
using ISynergy.Framework.Core.Services;
using ISynergy.Framework.OpenTelemetry.ApplicationInsights.Extensions;
using ISynergy.Framework.OpenTelemetry.Extensions;
using ISynergy.Framework.UI.Extensions;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.Logging;
using OpenTelemetry.Metrics;
using OpenTelemetry.Trace;
using {ApplicationName}.Extensions;
using {ApplicationName}.Services;
#if DEBUG
using Microsoft.Maui.DevFlow.Agent;
#endif

namespace {ApplicationName};

/// <summary>
/// Creates the MAUI application for every head. The platform entry points call <see cref="CreateMauiApp"/>.
/// </summary>
public static class MauiProgram
{
    private const string DefaultEnvironment = "Production";
    private const string DevelopmentEnvironment = "Development";

    /// <summary>
    /// Creates the application. The configuration comes from the embedded <c>appsettings</c> files. The environment is
    /// taken from the launch arguments, then from the Environment variable, then defaults to production.
    /// </summary>
    /// <returns>The configured application.</returns>
    /// <exception cref="InvalidOperationException">The main assembly or the base configuration cannot be loaded.</exception>
    public static MauiApp CreateMauiApp()
    {
        var builder = MauiApp.CreateBuilder();
        var mainAssembly = Assembly.GetAssembly(typeof(MauiProgram)) ?? throw new InvalidOperationException("The main assembly cannot be loaded.");

        var infoService = new InfoService();
        infoService.LoadAssembly(mainAssembly);

        var environment = ApplicationExtensions.TryGetEnvironmentFromLaunchArgs()
            ?? Environment.GetEnvironmentVariable("Environment")
            ?? DefaultEnvironment;

        builder.Configuration.AddConfiguration(LoadConfiguration(mainAssembly, environment));

        // The environment is read from configuration above, so the development check uses that value and not the host environment.
        var telemetry = builder.Logging.AddTelemetry(
            builder,
            infoService,
            tracerProviderBuilderAction: tracing => tracing.AddSource("Microsoft.Maui"),
            meterProviderBuilderAction: metrics => metrics.AddMeter("Microsoft.Maui"),
            loggerProviderBuilderAction: _ => { });

        if (string.Equals(environment, DevelopmentEnvironment, StringComparison.OrdinalIgnoreCase))
            telemetry.AddOtlpExporter();
        else
            telemetry.AddApplicationInsightsExporter();

        builder
            .UseMauiAppForCurrentHead<App>()
            // Registers the fonts the framework ships. The application adds none of its own.
            .UseFonts(_ => { })
            // The colours offered by the theme picker. The framework derives the full palette from the chosen colour.
            .UseThemeColors(colors =>
            {
                colors.Add("#512BD4");
                colors.Add("#0078D7");
                colors.Add("#107C10");
            })
            .ConfigureServices<App, Context, CommonServices, ExceptionHandlerService, SettingsService, global::{ApplicationName}.Resources.Properties.Resources>(
                appBuilder =>
                {
                    appBuilder.Add{ApplicationName}ApiServices();
// {{#if sqlite-device}}
                    appBuilder.Services.AddDeviceStore();
// {{/if}}
                },
                mainAssembly,
                assemblyName => assemblyName.Name!.StartsWith(typeof(App).Namespace!, StringComparison.Ordinal));

#if DEBUG
        builder.AddMauiDevFlowAgent();
#endif

        return builder.Build();
    }

    /// <summary>
    /// Reads the base configuration and, when one exists for the environment, the environment file on top of it.
    /// </summary>
    /// <param name="assembly">The assembly that embeds the configuration files.</param>
    /// <param name="environment">The environment name, for example <c>Development</c>.</param>
    /// <returns>The merged configuration.</returns>
    /// <exception cref="InvalidOperationException">The base configuration is not embedded.</exception>
    private static IConfiguration LoadConfiguration(Assembly assembly, string environment)
    {
        var configuration = new ConfigurationBuilder();

        var baseStream = assembly.GetManifestResourceStream("{ApplicationName}.appsettings.json")
            ?? throw new InvalidOperationException("The embedded base configuration is missing.");
        configuration.AddJsonStream(baseStream);

        var environmentStream = assembly.GetManifestResourceStream("{ApplicationName}.appsettings." + environment + ".json");
        if (environmentStream is not null)
            configuration.AddJsonStream(environmentStream);

        return configuration.Build();
    }
}
