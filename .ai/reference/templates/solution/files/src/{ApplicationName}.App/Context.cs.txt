using ISynergy.Framework.Core.Abstractions;
using ISynergy.Framework.Core.Constants;
using ISynergy.Framework.Core.Extensions;

namespace {ApplicationName};

/// <summary>
/// Holds the state of the signed-in user for the lifetime of the application. The framework reads it through
/// <see cref="IContext"/>. The sample has no sign-in, so the profile stays empty.
/// </summary>
public class Context : IContext
{
    /// <summary>Gets or sets the profile of the signed-in user, or null when nobody is signed in.</summary>
    public IProfile? Profile { get; set; }

    /// <summary>Gets the time zone of the profile, or the device time zone when there is no profile.</summary>
    public TimeZoneInfo TimeZone =>
        Profile is not null
            ? TimeZoneInfo.FindSystemTimeZoneById(Profile.TimeZoneId)
            : TimeZoneInfo.Local;

    /// <summary>Gets a value indicating whether a user is signed in.</summary>
    public bool IsAuthenticated => Profile is not null;

    /// <summary>Gets a value indicating whether the signed-in user is an administrator.</summary>
    public bool IsUserAdministrator =>
        Profile is not null && Profile.IsInRole(nameof(RoleNames.Administrator));

    /// <summary>Gets or sets a value indicating whether the device is offline.</summary>
    public bool IsOffline { get; set; }

    /// <summary>Gets the custom values that the application attaches to the context.</summary>
    public Dictionary<string, object> Properties { get; } = [];
}
