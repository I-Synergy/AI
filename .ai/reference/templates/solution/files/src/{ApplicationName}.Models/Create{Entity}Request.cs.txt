using System.ComponentModel.DataAnnotations;

namespace {ApplicationName}.Models;

/// <summary>
/// Request body that creates one {Entity}. The type is not positional, and its annotations sit on its
/// properties, so <c>.WithValidation&lt;Create{Entity}Request&gt;()</c> evaluates them.
/// </summary>
public sealed record Create{Entity}Request
{
    /// <summary>Gets the display name, required and at most 100 characters.</summary>
    [Required, MaxLength(100)]
    public string Name { get; init; } = string.Empty;

    /// <summary>Gets the optional description, at most 500 characters.</summary>
    [MaxLength(500)]
    public string? Description { get; init; }
}
