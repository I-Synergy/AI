using Microsoft.Data.Sqlite;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using {ApplicationName}.Abstractions.Services;
using {ApplicationName}.Data.Sqlite.Services;

namespace {ApplicationName}.Data.Sqlite.Extensions;

/// <summary>
/// Registers the device-local SQLite store.
/// </summary>
public static class ServiceCollectionExtensions
{
    /// <summary>
    /// Registers the store at <paramref name="databasePath"/> and creates the directory when it is missing. The schema
    /// is not created here: call <see cref="ISqliteLocalDataService.EnsureCreatedAsync"/> once at startup.
    /// </summary>
    /// <param name="services">The service collection to register into.</param>
    /// <param name="databasePath">The path of the SQLite file.</param>
    /// <returns>The same service collection, so calls can be chained.</returns>
    public static IServiceCollection AddLocalSqliteStore(this IServiceCollection services, string databasePath)
    {
        ArgumentNullException.ThrowIfNull(services);
        ArgumentException.ThrowIfNullOrWhiteSpace(databasePath);

        var fullPath = Path.GetFullPath(databasePath);
        var directory = Path.GetDirectoryName(fullPath);
        if (!string.IsNullOrEmpty(directory))
            Directory.CreateDirectory(directory);

        // The builder escapes the path, so a path with a semicolon or quote still forms a valid connection string.
        var connectionString = new SqliteConnectionStringBuilder { DataSource = fullPath }.ToString();
        services.AddDbContextFactory<LocalDataContext>(options => options.UseSqlite(connectionString));
        services.AddSingleton<ISqliteLocalDataService, SqliteLocalDataService>();

        return services;
    }
}
